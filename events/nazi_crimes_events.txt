        }

    }

        days = 3

            limit = { has_idea = ally_finish_GER_crime }
            swap_ideas = {
                remove_idea = ally_finish_GER_crime
                add_idea = ally_finish_GER_crime_2
            }
        }

            limit = {
                NOT = { 
                    has_idea = ally_finish_GER_crime
                    has_idea = ally_finish_GER_crime_2
                 }
            }
            add_ideas = ally_finish_GER_crime
        }
        add_war_support = 0.1
    }
    option = {
        name = nazi_crimes.100.b
        trigger = {
            NOT = { 
                has_government = fascism

            }
        }
